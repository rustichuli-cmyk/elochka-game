<script>
    // Инициализация Telegram WebApp
    const tg = window.Telegram.WebApp;
    tg.expand(); // Разворачиваем приложение на весь экран

    // Показываем ID пользователя из Telegram
    if (tg.initDataUnsafe && tg.initDataUnsafe.user) {
        document.getElementById('my-id').innerText = tg.initDataUnsafe.user.id;
    } else {
        document.getElementById('my-id').innerText = "1234567"; // Тестовый ID для ПК
    }

    // Состояние игры (пока локальное)
    let water = 5;
    let xp = 0;
    let level = 1;
    const treeStages = { 1: "🌱", 2: "🌿", 3: "🌲", 4: "🎄", 5: "🌟🎄🌟" };

    function switchScreen(screenId, btn) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
        btn.classList.add('active');
    }

    function waterTree() {
        if (water <= 0) {
            tg.showAlert("❌ Вода закончилась! Попросите друзей помочь.");
            return;
        }
        water--;
        xp += 25;

        if (xp >= 100) {
            level++;
            xp = 0;
            tg.HapticFeedback.notificationOccurred('success'); // Вибрация при повышении уровня
            document.getElementById('tree-art').innerText = treeStages[level] || "🌟🎄🌟";
        } else {
            tg.HapticFeedback.impactOccurred('light'); // Лёгкая вибрация при клике
        }

        updateUI();
    }

    function updateUI() {
        document.getElementById('water-txt').innerText = water;
        document.getElementById('level-txt').innerText = level;
        document.getElementById('xp-txt').innerText = `${xp} / 100 XP`;
        document.getElementById('progress').style.width = `${xp}%`;
    }

    function addFriend() {
        const id = document.getElementById('friend-id-input').value;
        if(!id) return;
        tg.showPopup({
            title: 'Добавление в друзья',
            message: `Вы хотите добавить пользователя с ID ${id}?`,
            buttons: [{type: 'ok', text: 'Да'}]
        });
    }

    updateUI();
</script>
